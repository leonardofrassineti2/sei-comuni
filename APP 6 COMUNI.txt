<script>
  const pulsanti = document.querySelectorAll("#filtri button");
  const sezioneNotizie = document.getElementById("notizie");

  pulsanti.forEach(pulsante => {
    pulsante.addEventListener("click", () => {
      pulsanti.forEach(p => p.classList.remove("attivo"));
      pulsante.classList.add("attivo");

      const comune = pulsante.dataset.comune;
      filtraNotizie(comune);
    });
  });

  function filtraNotizie(comune){
    const card = sezioneNotizie.querySelectorAll(".card");
    card.forEach(elemento => {
      const etichetta = elemento.querySelector(".etichetta");
      if(!etichetta) return;

      const testo = etichetta.textContent.toLowerCase();

      if(comune === "tutti"){
        elemento.style.display = "";
        return;
      }

      const nomi = {
        "riolo-terme": "riolo terme",
        "faenza": "faenza",
        "castel-bolognese": "castel bolognese",
        "casola-valsenio": "casola valsenio",
        "brisighella": "brisighella",
        "solarolo": "solarolo"
      };

      elemento.style.display = testo.includes(nomi[comune]) ? "" : "none";
    });
  }

  function aggiorna(){
    alert("L’aggiornamento automatico verrà attivato dopo il collegamento delle fonti dei Comuni.");
  }
</script>
